### Part 2: AI Content Generation (15 min)

**Generate Blog Post with AI:**
```php
<?php
// File: wp-content/plugins/ai-content/includes/class-content-generator.php

class AI_Content_Generator {

    private $ai_client;

    public function __construct() {
        $this->ai_client = new WP_AI_Client();
    }

    public function generate_blog_post($topic, $keywords = array(), $tone = 'professional') {
        $prompt = $this->build_prompt($topic, $keywords, $tone);

        $content = $this->ai_client->generate_text($prompt, array(
            'max_tokens' => 2000,
            'temperature' => 0.7,
        ));

        return array(
            'title'   => $this->extract_title($content),
            'content' => $this->extract_content($content),
            'excerpt' => $this->extract_excerpt($content),
            'tags'    => $this->extract_tags($content),
        );
    }

    private function build_prompt($topic, $keywords, $tone) {
        $keywords_str = !empty($keywords) ? implode(', ', $keywords) : 'none';

        return <<<PROMPT
Write a comprehensive blog post about: {$topic}

Requirements:
- Target keywords: {$keywords_str}
- Tone: {$tone}
- Length: 1000-1500 words
- Include H2 and H3 headings
- Add practical examples
- Conclude with actionable takeaways

Format:
[TITLE]Your engaging title here[/TITLE]
[CONTENT]Your full blog post content here[/CONTENT]
[EXCERPT]A compelling 2-sentence summary[/EXCERPT]
[TAGS]tag1, tag2, tag3, tag4, tag5[/TAGS]
PROMPT;
    }

    private function extract_title($content) {
        preg_match('/\[TITLE\](.*?)\[\/TITLE\]/s', $content, $matches);
        return !empty($matches[1]) ? trim($matches[1]) : 'Untitled Post';
    }

    private function extract_content($content) {
        preg_match('/\[CONTENT\](.*?)\[\/CONTENT\]/s', $content, $matches);
        return !empty($matches[1]) ? trim($matches[1]) : '';
    }

    private function extract_excerpt($content) {
        preg_match('/\[EXCERPT\](.*?)\[\/EXCERPT\]/s', $content, $matches);
        return !empty($matches[1]) ? trim($matches[1]) : '';
    }

    private function extract_tags($content) {
        preg_match('/\[TAGS\](.*?)\[\/TAGS\]/s', $content, $matches);
        if (!empty($matches[1])) {
            return array_map('trim', explode(',', $matches[1]));
        }
        return array();
    }

    public function create_draft_post($topic, $keywords = array()) {
        $generated = $this->generate_blog_post($topic, $keywords);

        $post_id = wp_insert_post(array(
            'post_title'   => $generated['title'],
            'post_content' => $generated['content'],
            'post_excerpt' => $generated['excerpt'],
            'post_status'  => 'draft',  // Requires human review
            'post_type'    => 'post',
        ));

        if ($post_id && !empty($generated['tags'])) {
            wp_set_post_tags($post_id, $generated['tags']);
        }

        return $post_id;
    }
}
```

### Part 3: SEO Optimization with AI (10 min)

**Optimize Existing Content:**
```php
<?php
class AI_SEO_Optimizer {

    private $ai_client;

    public function __construct() {
        $this->ai_client = new WP_AI_Client();
    }

    public function optimize_for_seo($post_id, $target_keyword) {
        $post = get_post($post_id);

        if (!$post) {
            return new WP_Error('invalid_post', 'Post not found');
        }

        $prompt = <<<PROMPT
Optimize this blog post for SEO targeting the keyword: "{$target_keyword}"

Current Title: {$post->post_title}
Current Content: {$post->post_content}

Provide:
1. Optimized title (60 chars max, include keyword)
2. Meta description (155 chars max, compelling, include keyword)
3. Suggested internal link anchor text (3-5 suggestions)
4. Readability improvements (list of specific suggestions)
5. Keyword density analysis

Format as JSON.
PROMPT;

        $response = $this->ai_client->generate_text($prompt);
        $suggestions = json_decode($response, true);

        return array(
            'optimized_title'      => $suggestions['title'] ?? $post->post_title,
            'meta_description'     => $suggestions['meta_description'] ?? '',
            'internal_links'       => $suggestions['internal_links'] ?? array(),
            'readability_tips'     => $suggestions['readability'] ?? array(),
            'keyword_density'      => $suggestions['keyword_density'] ?? 'N/A',
        );
    }

    public function suggest_related_content($post_id, $count = 5) {
        $post = get_post($post_id);

        $prompt = <<<PROMPT
Based on this blog post:
Title: {$post->post_title}
Excerpt: {$post->post_excerpt}

Suggest {$count} related blog post topics that would complement this content.
Provide as JSON array of objects with: title, keywords (array), estimated_word_count
PROMPT;

        $response = $this->ai_client->generate_text($prompt);
        return json_decode($response, true);
    }
}
```

### Part 4: Quality Control (10 min)

**Implement Content Review:**
```php
<?php
class AI_Content_Quality_Control {

    private $ai_client;

    public function __construct() {
        $this->ai_client = new WP_AI_Client();
    }

    public function review_ai_generated_content($content) {
        $prompt = <<<PROMPT
Review this AI-generated content for quality issues:

{$content}

Check for:
1. Factual accuracy concerns
2. Grammatical errors
3. Tone consistency
4. Potential plagiarism indicators
5. Missing citations
6. Logical flow issues
7. Overall quality score (1-10)

Provide detailed feedback as JSON.
PROMPT;

        $review = $this->ai_client->generate_text($prompt);
        return json_decode($review, true);
    }

    public function is_publishable($review_data) {
        $quality_score = $review_data['quality_score'] ?? 0;

        // Require quality score >= 7 to publish
        if ($quality_score < 7) {
            return false;
        }

        // Require no factual accuracy concerns
        if (!empty($review_data['factual_concerns'])) {
            return false;
        }

        // Require no plagiarism indicators
        if (!empty($review_data['plagiarism_indicators'])) {
            return false;
        }

        return true;
    }
}
```

**Task:** Launch multiple Claude subagents to execute WordPress tasks in parallel, dramatically reducing total completion time.

**Context:** Subagents are independent Claude instances that can work on different tasks simultaneously. Instead of doing 5 tasks sequentially (50 minutes), you can do them in parallel (10 minutes).

### Requirements

1. Identify tasks suitable for parallel execution
2. Launch 3+ subagents simultaneously
3. Coordinate their work
4. Aggregate results from all subagents
5. Document use cases for subagents

### Part 1: Understand When to Use Subagents (10 min)

**✅ Good Use Cases for Subagents:**
- Creating multiple similar plugins
- Testing across multiple environments
- Generating content for multiple pages
- Analyzing multiple codebases
- Running multiple tests simultaneously

**❌ Bad Use Cases:**
- Tasks that depend on each other
- Single linear workflow
- Shared state modifications
- Database migrations (risk of conflicts)

**Example Scenario:**
You need to create 3 custom Gutenberg blocks for a client:
- Hero block
- Testimonial block
- CTA block

**Sequential (Traditional):** 15 min + 15 min + 15 min = 45 minutes
**Parallel (Subagents):** max(15, 15, 15) = 15 minutes

### Part 3: AI Integration (15 min)

**Backend AJAX Handler:**
```php
<?php
// File: wp-content/plugins/ai-content-suggestions/includes/class-ajax-handler.php

class AI_Suggestions_AJAX_Handler {

    private $ai_client;

    public function __construct() {
        $this->ai_client = new WP_AI_Client();
        add_action('wp_ajax_get_ai_suggestions', array($this, 'get_suggestions'));
    }

    public function get_suggestions() {
        check_ajax_referer('ai_suggestions_nonce', 'nonce');

        $title = sanitize_text_field($_POST['title']);
        $content = wp_kses_post($_POST['content']);

        if (empty($content)) {
            wp_send_json_error('No content provided');
        }

        try {
            $suggestions = $this->generate_suggestions($title, $content);
            wp_send_json_success($suggestions);

        } catch (Exception $e) {
            wp_send_json_error($e->getMessage());
        }
    }

    private function generate_suggestions($title, $content) {
        $prompt = <<<PROMPT
Analyze this blog post and provide content suggestions:

Title: {$title}
Content: {$content}

Provide:
1. 5 related topics to cover
2. 3-5 internal link opportunities (suggest anchor text and relevant existing pages)
3. SEO score (0-100) based on:
   - Keyword usage
   - Content length
   - Heading structure
   - Readability
4. SEO feedback (2-3 specific improvements)

Return as JSON with keys: topics (array), internal_links (array of objects), seo_score (number), seo_feedback (string)
PROMPT;

        $response = $this->ai_client->generate_text($prompt, array(
            'max_tokens' => 1500,
            'temperature' => 0.6,
        ));

        $data = json_decode($response, true);

        if (json_last_error() !== JSON_ERROR_NONE) {
            throw new Exception('Failed to parse AI response');
        }

        // Find actual internal link URLs
        $data['internal_links'] = $this->find_internal_links($data['internal_links']);

        return $data;
    }

    private function find_internal_links($suggested_links) {
        $links_with_urls = array();

        foreach ($suggested_links as $link) {
            $search_query = $link['search_term'] ?? $link['title'];

            // Find matching post
            $posts = get_posts(array(
                's'              => $search_query,
                'posts_per_page' => 1,
                'post_status'    => 'publish',
            ));

            if (!empty($posts)) {
                $links_with_urls[] = array(
                    'title'       => $posts[0]->post_title,
                    'url'         => get_permalink($posts[0]),
                    'anchor_text' => $link['anchor_text'],
                );
            }
        }

        return $links_with_urls;
    }
}

new AI_Suggestions_AJAX_Handler();
```

### Part 4: Error Handling & Fallbacks (5 min)

**Add Robust Error Handling:**
```php
<?php
private function generate_suggestions($title, $content) {
    try {
        $response = $this->ai_client->generate_text($prompt, array(
            'max_tokens' => 1500,
            'temperature' => 0.6,
            'timeout'     => 15,  // 15 second timeout
        ));

        $data = json_decode($response, true);

        if (json_last_error() !== JSON_ERROR_NONE) {
            // Fallback: Return basic suggestions
            return $this->get_fallback_suggestions($title, $content);
        }

        return $data;

    } catch (Exception $e) {
        error_log('AI Suggestions Error: ' . $e->getMessage());

        // Fallback to rule-based suggestions
        return $this->get_fallback_suggestions($title, $content);
    }
}

private function get_fallback_suggestions($title, $content) {
    // Non-AI fallback suggestions
    return array(
        'topics' => array(
            'Expand on the introduction',
            'Add practical examples',
            'Include case studies',
            'Add actionable takeaways',
        ),
        'internal_links' => $this->find_related_posts($title),
        'seo_score' => $this->calculate_basic_seo_score($content),
        'seo_feedback' => 'Add more headings and improve content structure.',
    );
}
```

- Code scaffolding
